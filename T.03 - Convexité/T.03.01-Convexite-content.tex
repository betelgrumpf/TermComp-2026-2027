\section{Dérivée seconde}

\begin{bdefi}
	Soit une fonction \(f\) dérivable sur un intervalle \(I\) dont la dérivée \(f^{\prime}\) est dérivable sur \(I\).\\
	On appelle fonction dérivée seconde de \(f\) sur \(I\) la dérivée de \(f^{\prime}\) et on note :\\
	\(f^{\prime \prime}(x)=\left(f^{\prime}(x)\right)^{\prime}\).
\end{bdefi}

\vspace*{-0.3cm}
\begin{bmethodet}{Calculer la dérivée seconde d'une fonction}
	\grasInMet{Enoncé :}
	
	Calculer la dérivée seconde de chacune des fonctions \(f\) et \(g\) définies par :
	\begin{enumMet}
		\item \(f(x)=3 x^{3}-5 x^{2}+1\)
		\item \(g(x)=x e^{x}\)
	\end{enumMet}
	
	\grasInMet{Réponse :}
	
	\begin{enumMet}
		\item \(f^{\prime}(x)=9 x^{2}-10 x\)\\
		\(f^{\prime \prime}(x)=\left(f^{\prime}(x)\right)^{\prime}=18 x-10\)
		\item \(g^{\prime}(x)=1 \times e^{x}+x e^{x}=e^{x}(1+x)\)\\
		\(g^{\prime \prime}(x)=e^{x}(1+x)+e^{x} \times 1=e^{x}(2+x)\)
	\end{enumMet}
\end{bmethodet}

\section{Fonction convexe et fonction concave}

\subsection{Définitions avec les cordes}

\begin{bdefi}
	Une corde est un segment reliant deux points d'une courbe.
	\begin{center}
		\includegraphics[scale=0.2]{Img-Cours-01}
	\end{center}
\end{bdefi}


\begin{bdefis}
	Soit une fonction \(f\) définie sur un intervalle \(I\).
	\begin{itemDef}
		\item La fonction \(f\) est convexe sur \(I\), si sa courbe est entièrement située en dessous de chacune de ses cordes.
		\item La fonction \(f\) est concave sur \(I\), si sa courbe est entièrement située au-dessus de chacune de ses cordes.
	\end{itemDef}
\end{bdefis}

\subsection{Propriétés}

\begin{bprops}
	\begin{itemProp}
		\item La fonction carré \(x \mapsto x^{2}\) est convexe sur \(\mathbb{R}\).
		\item La fonction cube \(x \longmapsto x^{3}\) est concave sur \(\left.]-\infty ; 0\right]\) et convexe sur \([0 ;+\infty[\).
		\item La fonction inverse \(x \mapsto \frac{1}{x}\) est concave sur \(]-\infty ; 0[\) et convexe sur \(] 0 ;+\infty[\).
		\item La fonction racine carrée \(x \longmapsto \sqrt{x}\) est concave sur \([0 ;+\infty[\).
	\end{itemProp}
\end{bprops}

\begin{bprop}
	Soit une fonction \(f\) définie et deux fois dérivable sur un intervalle \(I\).
	
	\begin{itemProp}
		\item Dire que la fonction \(f\) est convexe sur \(I\), revient à dire que sa dérivée \(f^{\prime}\) est croissante sur \(I\), soit : \(f^{\prime \prime}(x) \geq 0\).
		\item Dire que la fonction \(f\) est concave sur \(I\), revient à dire que sa dérivée \(f^{\prime}\) est décroissante sur \(I\), soit : \(f^{\prime \prime}(x) \leq 0\).
	\end{itemProp}
\end{bprop}		

\begin{brmq}
	Dans la pratique, pour étudier la convexité d'une fonction, on détermine le signe de la dérivée seconde.
\end{brmq}

\begin{bmethodet}{Étudier la convexité d'une fonction}
	\grasInMet{Enoncé :}
	
	Soit la fonction \(f\) définie sur \(\mathbb{R}\) par \(f(x)=\frac{1}{3} x^{3}-9 x^{2}+4\).\\
	Étudier la convexité de la fonction \(f\).
	
	\grasInMet{Correction : }
	
	\(f^{\prime}(x)=x^{2}-18 x\).\\
	\(f^{\prime \prime}(x)=2 x-18\)\\
	On a: \(f^{\prime \prime}(x)=0\) pour \(x=9\).
	
	Pour tout \(x \leq 9, f^{\prime \prime}(x) \leq 0\).\\
	Pour tout \(x \geq 9, f^{\prime \prime}(x) \geq 0\).
	
	Donc \(f\) est concave sur \(]-\infty ; 9]\) et \(f\) est convexe sur \([9 ;+\infty[\).
\end{bmethodet}

\begin{bdefi}
	\begin{minipage}[c]{0.48\linewidth}
		Soit une fonction \(f\) dérivable sur un intervalle \(I\).\\
		Un point d'inflexion est un point où la courbe traverse sa tangente.\\
	\end{minipage} \hfill
	\begin{minipage}[c]{0.48\linewidth}
		\includegraphics[scale=0.2]{Img-Cours-10}
	\end{minipage}
\end{bdefi}

\begin{bprop}
	Au point d'inflexion, la fonction change de convexité.
\end{bprop}

\begin{bexemple}
	
	\begin{minipage}[c]{0.48\linewidth}
		On considère la fonction cube \(x \mapsto x^{3}\).\\
		La tangente en 0 est l'axe des abscisses.\\
		Pour \(x \leq 0\), la courbe est en dessous de sa tangente.\\
		Pour \(x \geq 0\), la courbe est au-dessus de sa tangente.\\
		L'origine est donc le point d'inflexion de la courbe de la fonction cube.\\
		La tangente à la courbe traverse donc la courbe en ce point.\\
	\end{minipage} \hfill
	\begin{minipage}[c]{0.48\linewidth}
		\includegraphics[scale=0.2]{Img-Cours-11}
	\end{minipage}
\end{bexemple}

\begin{bmethodet}{Étudier la convexité pour résoudre un problème}
	\grasInMet{Enoncé :}
	
	Une entreprise fabrique des clés USB avec un maximum de 10000 par mois. Le coût de fabrication \(C\) (en milliers d'euros) de \(x\) milliers de clés produites s'exprime par : \(C(x)=0,05 x^{3}-1,05 x^{2}+8 x+4\), définie sur l'intervalle [ \(0 ; 10\) ].
	
	\begin{enumMet}
		\item À l'aide de la calculatrice, conjecturer la convexité de la fonction \(C\).
		
		En déduire si la courbe possède un point d'inflexion.
		\item Démontrer ces résultats.
		\item Interpréter les résultats obtenus au regard du contexte de l'exercice.
	\end{enumMet}
	
	
	
	\grasInMet{Correction : }
	
	\begin{enumMet}
		\item La fonction semble concave sur l'intervalle [ 0 ; 7] et convexe sur l'intervalle [7; 10]. La courbe semble posséder un point d'inflexion pour \(x=7\).\\
		\includegraphics[scale=0.55]{Img-Cours-16-2}
		\item \(C(x)=0,05 x^{3}-1,05 x^{2}+8 x+4\)
	
	
	\[
	\begin{aligned}
		& C^{\prime}(x)=0,15 x^{2}-2,1 x+8 \\
		& C^{\prime \prime}(x)=0,3 x-2,1
	\end{aligned}
	\]
	
	Or, \(0,3 x-2,1=0\) pour \(x=7\).\\
	On peut ainsi résumer les variations de \(C^{\prime}\) et la convexité de \(C\) dans le tableau suivant :
	
	\includegraphics[scale=0.7]{Img-Cours-17}
	
	\(C(7)=25,7\)\\
	Ainsi, le point de coordonnées ( \(7 ; 25,7\) ) est un point d'inflexion de la courbe.\\
	\item Avant le point d'inflexion, la fonction est concave, la croissance du coût de fabrication \(C\) ralentie. Après le point d'inflexion, la fonction est convexe, la croissance du coût de fabrication s'accélère.\\
	Ainsi, à partir de 7000 clés produites, la croissance du coût de fabrication s'accélère.
	\end{enumMet}
\end{bmethodet}


\begin{bmethodet}{Étudier la convexité pour résoudre un problème}
	\grasInMet{Enoncé :}
	
	Soit \(f\) la fonction définie sur \(\mathbb{R}\) par \(f(x)=x e^{-\frac{x}{2}}\).\\
	\begin{enumMet}
		\item Calculer la dérivée de la fonction \(f\).
		\item Dresser le tableau de variations de la fonction \(f\).
		\item Tracer la courbe représentative de la fonction \(f\) en s'aidant de la calculatrice.
		\item Déterminer une valeur approchée de l'abscisse du point d'inflexion à la courbe.
		\item Démontrer que \(f^{\prime \prime}(x)=\left(\frac{x}{4}-1\right) e^{-\frac{x}{2}}\).
		\item En déduire l'abscisse du point d'inflexion.
	\end{enumMet}
	
	
	
	\grasInMet{Correction : }
	
	\begin{enumMet}
		\item \(f^{\prime}(x)=e^{-\frac{x}{2}}+x \times\left(-\frac{1}{2}\right) e^{-\frac{x}{2}}=\left(1-\frac{x}{2}\right) e^{-\frac{x}{2}}\).
		\item Comme \(e^{-\frac{x}{2}}>0, f^{\prime}(x)\) est du signe de \(1-\frac{x}{2}\). \(f\) est donc croissante sur l'intervalle \(]-\infty ; 2]\) et décroissante sur l'intervalle \([2 ;+\infty[\).
		
		On dresse le tableau de variations :
		
		\includegraphics[scale=0.7]{Img-Cours-18}
		
		\(f(2)=2 e^{-\frac{2}{2}}=2 e^{-1}=\dfrac{2}{e}\)\\
		\item \ \\
		\includegraphics[scale=0.2]{Img-Cours-19}\\
		\item Le point d'inflexion semble avoir pour abscisse une valeur proche de 4 .\\
		\item \(f^{\prime \prime}(x)=-\frac{1}{2} e^{-\frac{x}{2}}+\left(-\frac{1}{2}\right)\left(1-\frac{x}{2}\right) e^{-\frac{x}{2}}\)
		
		\[
		\begin{aligned}
			& =-\frac{1}{2} e^{-\frac{x}{2}}-\dfrac{1}{2} e^{-\frac{x}{2}}+\frac{x}{4} e^{-\frac{x}{2}} \\
			& =-e^{-\frac{x}{2}}+\frac{x}{4} e^{-\frac{x}{2}} \\
			& =\left(\frac{x}{4}-1\right) e^{-\frac{x}{2}}
		\end{aligned}
		\]
		
		\item Comme \(e^{-\frac{x}{2}}>0, f^{\prime \prime}(x)\) est du signe de \(\frac{x}{4}-1\).
		
		Donc \(f^{\prime \prime}(x) \geq 0\) pour \(\frac{x}{4}-1 \geq 0\) soit \(x \geq 4\).\\
		\(f^{\prime \prime}(x) \leq 0\) pour \(\frac{x}{4}-1 \leq 0\) soit \(x \leq 4\).\\
		Ainsi \(f^{\prime}\) est croissante sur \([4 ;+\infty[\) et donc \(f\) est convexe sur cet intervalle.\\
		\(f^{\prime}\) est décroissante sur \(\left.]-\infty ; 4\right]\) et donc \(f\) est concave sur cet intervalle.\\
		On en déduit que la courbe représentative de \(f\) possède un point d'inflexion d'abscisse 4.
	\end{enumMet}
\end{bmethodet}
