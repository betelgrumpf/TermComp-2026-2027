\section*{Exercice 7 -- Vrai ou Faux}

\textbf{1) FAUX.} La fonction $f$ n'est pas définie sur $[-5; 20]$ mais sur $[0; 15]$.

\textbf{2) VRAI.} Sur $[0; 5]$, $f'$ est décroissante, donc $f''(x) \leq 0$, donc $f$ est concave.

\textbf{3) VRAI.} En $x = 5$, $f'$ passe d'une phase décroissante à croissante, donc $f''$ change de signe : il y a un point d'inflexion.

\textbf{4) FAUX.} $f'(x) = 0$ a deux solutions (en $x = 0$ et entre 5 et 15), donc il y a deux tangentes horizontales.

\textbf{5) VRAI.} Sur $[5; 15]$, $f'$ est croissante, donc $f''(x) \geq 0$, donc $f''$ est positive.

\section*{Exercice 8}

Soit $f(x) = 0,5x^2 + 2x - 1$ et $g(x) = -0,5x^2 - x + 3,5$

\textbf{1) Conjecture}

\begin{itemize}
	\item $f$ semble \textbf{convexe} sur $\mathbb{R}$ (parabole tournée vers le haut)
	\item $g$ semble \textbf{concave} sur $\mathbb{R}$ (parabole tournée vers le bas)
\end{itemize}

\textbf{2) Calculs des dérivées}

a) $f'(x) = x + 2$ \quad et \quad $f''(x) = 1$

b) $g'(x) = -x - 1$ \quad et \quad $g''(x) = -1$

\textbf{3) Démonstration}

\begin{itemize}
	\item $f''(x) = 1 > 0$ pour tout $x \in \mathbb{R}$, donc \textbf{$f$ est convexe sur $\mathbb{R}$}.
	\item $g''(x) = -1 < 0$ pour tout $x \in \mathbb{R}$, donc \textbf{$g$ est concave sur $\mathbb{R}$}.
\end{itemize}

\section*{Exercice 9}

\textbf{1)} $f(x) = x^4 - 3x + 2$

$f'(x) = 4x^3 - 3$ \quad ; \quad $f''(x) = 12x^2 \geq 0$ pour tout $x \in \mathbb{R}$

$\Rightarrow$ \textbf{$f$ est convexe sur $\mathbb{R}$}

\textbf{2)} $f(x) = 10e^{-0,5x} + 2$

$f'(x) = -5e^{-0,5x}$ \quad ; \quad $f''(x) = 2,5e^{-0,5x} > 0$ pour tout $x \in \mathbb{R}$

$\Rightarrow$ \textbf{$f$ est convexe sur $\mathbb{R}$}

\textbf{3)} $f(x) = -2x^2 + 3$

$f'(x) = -4x$ \quad ; \quad $f''(x) = -4 < 0$ pour tout $x \in \mathbb{R}$

$\Rightarrow$ \textbf{$f$ est concave sur $\mathbb{R}$}

\textbf{4)} $f(x) = -5e^{3x} + 4$

$f'(x) = -15e^{3x}$ \quad ; \quad $f''(x) = -45e^{3x} < 0$ pour tout $x \in \mathbb{R}$

$\Rightarrow$ \textbf{$f$ est concave sur $\mathbb{R}$}

\section*{Exercice 10}

Les points d'inflexion sont les points où $f''(x) = 0$ avec changement de signe.

\textbf{1)} $f(x) = x^3 - 3x + 5$

$f'(x) = 3x^2 - 3$ \quad ; \quad $f''(x) = 6x$

$f''(x) = 0 \Leftrightarrow x = 0$ (changement de signe de $-$ à $+$)

$\Rightarrow$ \textbf{Point d'inflexion : $(0; 5)$}

\textbf{2)} $f(x) = e^x - \dfrac{1}{2}x^2$

$f'(x) = e^x - x$ \quad ; \quad $f''(x) = e^x - 1$

$f''(x) = 0 \Leftrightarrow e^x = 1 \Leftrightarrow x = 0$ (changement de signe)

$f(0) = 1$

$\Rightarrow$ \textbf{Point d'inflexion : $(0; 1)$}

\textbf{3)} $f(x) = (x-1)^3 + 2$

$f'(x) = 3(x-1)^2$ \quad ; \quad $f''(x) = 6(x-1)$

$f''(x) = 0 \Leftrightarrow x = 1$ (changement de signe de $-$ à $+$)

$f(1) = 2$

$\Rightarrow$ \textbf{Point d'inflexion : $(1; 2)$}

\textbf{4)} $f(x) = \dfrac{1}{x^2+1}$

$f'(x) = \dfrac{-2x}{(x^2+1)^2}$ \quad ; \quad $f''(x) = \dfrac{-2(x^2+1)^2 + 8x^2(x^2+1)}{(x^2+1)^4} = \dfrac{6x^2-2}{(x^2+1)^3}$

$f''(x) = 0 \Leftrightarrow 6x^2 = 2 \Leftrightarrow x = \pm\dfrac{1}{\sqrt{3}}$

$f\left(\dfrac{1}{\sqrt{3}}\right) = \dfrac{3}{4}$

$\Rightarrow$ \textbf{Points d'inflexion : $\left(-\dfrac{1}{\sqrt{3}}; \dfrac{3}{4}\right)$ et $\left(\dfrac{1}{\sqrt{3}}; \dfrac{3}{4}\right)$}

\textbf{5)} $f(x) = e^{-x^2}$

$f'(x) = -2xe^{-x^2}$ \quad ; \quad $f''(x) = -2e^{-x^2} + 4x^2e^{-x^2} = e^{-x^2}(4x^2-2)$

$f''(x) = 0 \Leftrightarrow 4x^2 = 2 \Leftrightarrow x = \pm\dfrac{1}{\sqrt{2}}$

$f\left(\dfrac{1}{\sqrt{2}}\right) = e^{-1/2}$

$\Rightarrow$ \textbf{Points d'inflexion : $\left(-\dfrac{1}{\sqrt{2}}; e^{-1/2}\right)$ et $\left(\dfrac{1}{\sqrt{2}}; e^{-1/2}\right)$}

\textbf{6)} $f(x) = xe^{-x}$

$f'(x) = e^{-x} - xe^{-x} = e^{-x}(1-x)$ \quad ; \quad $f''(x) = -e^{-x}(1-x) - e^{-x} = e^{-x}(x-2)$

$f''(x) = 0 \Leftrightarrow x = 2$ (changement de signe de $-$ à $+$)

$f(2) = 2e^{-2}$

$\Rightarrow$ \textbf{Point d'inflexion : $(2; 2e^{-2})$}
