\exerciceDS{ $-$ Vrai ou Faux}
Soit $f$ une fonction dérivable sur $[0 ; 15]$ pour laquelle on connaît le tableau de variations de la dérivée $f^{\prime}$.

\begin{center}
	\begin{tikzpicture}
		\tkzTabInit{$x$ / 1 , $f'(x)$ / 1.5}{$0$, $5$, $15$}
		\tkzTabVar{+/ $30$, -/ $-5$, +/ $20$}
	\end{tikzpicture}
\end{center}


Préciser si les affirmations suivantes sont vraies ou fausses.

\begin{enumMet}
	\item La fonction $f$ est convexe sur $[-5 ; 20]$.
	\item La fonction $f$ est concave sur $[0 ; 5]$.
	\item La courbe $C_{f}$ admet un point d'inflexion en 5 .
	\item La courbe $C_{f}$ admet une unique tangente parallèle à l'axe des abscisses.
	\item La fonction $f^{\prime \prime}$ est positive sur $[5 ; 15]$.
\end{enumMet}

\exerciceDS{}
Soient les fonctions $f$ et $g$ définies sur $\mathbb{R}$ par:

$$
\begin{gathered}
	f(x)=0,5 x^{2}+2 x-1 \\
	\text { et } g(x)=-0,5 x^{2}-x+3,5
\end{gathered}
$$

\begin{enumMet}
	\item On a obtenu à la calculatrice les courbes représentatives de $f$ et $g$ ci-contre. Après les avoir repérées, conjecturer si $f$, puis $g$, est convexe ou concave sur $\mathbb{R}$.
	\begin{center}
		\includegraphics[scale=0.25]{2025_11_06_3b44234b8b55fbc3078dg-4}
	\end{center}
	
	\item \begin{enumMet}
		\item Calculer $f^{\prime}(x)$, puis $f^{\prime \prime}(x)$.
		\item Calculer $g^{\prime}(x)$, puis $g^{\prime \prime}(x)$.
	\end{enumMet}
	\item Démontrer les conjectures émises.
\end{enumMet}
