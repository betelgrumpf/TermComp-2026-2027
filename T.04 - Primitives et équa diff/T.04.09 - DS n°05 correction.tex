\exerciceDS{$-$ (4,5 points)}

Dans chaque cas, déterminer la primitive $F$ de la fonction $f$ vérifiant la condition proposée.
%\begin{multicols}{2}
	\begin{enumMet}
		\item $f(x)=5 x-2\mathrm{e}^{-x}$, pour tout $x \in \mathbb{R}$ et $F(0)=4$.
		\begin{corrBox}
			$F(x)=\dfrac{5}{2}x^2+2e^{-x}+C$
			
			De plus $F(0)=4 \iff 4=\dfrac{5}{2}\times 0^2+2e^{0}+C\iff 4=2+C\iff C=2$
			
			Donc $F(x)=\dfrac{5}{2}x^2+2e^{-x}+2$
		\end{corrBox}
		\item $f(x)=8x \mathrm{e}^{2x^2}$, pour tout $x \in \mathbb{R}$ et $F(0)=5$.
		\begin{corrBox}
				$F(x)=2e^{2x^2}+C$
				
				De plus $F(0)=5 \iff 5=2e^{0}+C\iff 5=2+C\iff C=3$
				
				Donc $F(x)=2e^{2x^2}+3$
		\end{corrBox}
		\item $f(x)=\dfrac{10x-2}{5 x^2-2x+2}$ pour tout $x \in \mathbb{R}$ et $F(0)=\ln 6$.
		\begin{corrBox}
			$f(x)$ est du type $\dfrac{u'}{u}$ donc $F$ est du type $\ln(u)$
			
			$F(x)=\ln(5x^2-2x+2)+C$
			
			De plus $F(0)=\ln 6 \iff \ln 6=\ln 2+C\iff C=\ln 6 - \ln 2 = \ln 3$		
			
			$F(x)=\ln(5x^2-2x+2)+\ln 3$
		\end{corrBox}
	\end{enumMet}

\exerciceDS{ $-$ (3,5 points)}

Déterminer les solutions des équations différentielles suivantes.
%\begin{multicols}{3}
	\begin{enumMet}
		\item \ding{46} $y'=2e^x$
		\begin{corrBox}
			Les solutions de l'équation sont les primitives de $2e^x$ donc : $f(x)=2e^x+C$ avec C une constante réelle.
		\end{corrBox}
		\item $y^{\prime}=4 e^{2 x}$ et $y(0)=5$
		\begin{corrBox}
			Les solutions de l'équation sont les primitives de $4 e^{2 x}$ donc : $f(x)=2e^{2x}+C$ avec C une constante réelle.
			
			De plus $y(0)=5$ donc $2e^{2\times 0}+C=5$ d'ou $C=3$.
			
			La solution de l'équation différentielle est donc $f(x)=2e^{2x}+3$
		\end{corrBox}
		\item $y'=3x+4$
		\begin{corrBox}
			Les solutions de l'équation sont les primitives de $3x+4$ donc : $f(x)=\dfrac{3}{2}x^2+4x+C$ avec C une constante réelle.
		\end{corrBox}
	\end{enumMet}

		\end{corrBox}
