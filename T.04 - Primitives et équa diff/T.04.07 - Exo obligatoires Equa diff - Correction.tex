\exerciceDS{}

Déterminer les solutions des équations différentielles suivantes.
%\begin{multicols}{3}
	\begin{enumMet}
	\item $y'=2e^x$
	\begin{corrBox}
		Les solutions de l'équation sont les primitives de $2e^x$ donc : $f(x)=2e^x+C$ avec C une constante réelle.
	\end{corrBox}
	\item $y'=\dfrac{1}{t^2}$
	\begin{corrBox}
		Les solutions de l'équation sont les primitives de $\dfrac{1}{t^2}$ donc : $f(x)=-\dfrac{1}{t}+C$ avec C une constante réelle.
	\end{corrBox}
	\item $y^{\prime}=3 e^{6 x}$ et $y(0)=4$
	\begin{corrBox}
		Les solutions de l'équation sont les primitives de $3 e^{6 x}$ donc : $f(x)=\dfrac{3}{6}e^{6x}+C=\dfrac{1}{2}e^{6x}+C$ avec C une constante réelle.
		
		De plus $y(0)=4$ donc $\dfrac{1}{2}e^{6\times 0}+C=4$ d'ou $C=3,5$.
		
		La solution de l'équation différentielle est donc $f(x)=\dfrac{1}{2}e^{6x}+3,5$
	\end{corrBox}
	\item $y'=e^{-x}$
	\begin{corrBox}
		Les solutions de l'équation sont les primitives de $e^{-x}$ donc : $f(x)=-e^{-x}+C$ avec C une constante réelle.
	\end{corrBox}
\end{enumMet}
%\end{multicols}

\exerciceDS{ $-$ Équation différentielle $y^{\prime}=a y$ et conditions initiales}

Résoudre les équations différentielles suivantes.
%\begin{multicols}{3}
	\begin{enumMet}
	\item $y^{\prime}=y$ et $y(0)=5$
	\begin{corrBox}
		L'équation est du type $y'=ay$ avec $a=1$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{x}$ avec C une constante réelle.	
		
		De plus $y(0)=5$ donc $Ce^{0}=5$ d'ou $C=5$.
		
		La solution de l'équation différentielle est donc $f(x)=5e^{x}$	
	\end{corrBox}
	\item $y^{\prime}=2 y$ et $y(0)=4$
	\begin{corrBox}
		L'équation est du type $y'=ay$ avec $a=2$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{2x}$ avec C une constante réelle.	
		
		De plus $y(0)=4$ donc $Ce^{2\times 0}=4$ d'ou $C=4$.
		
		La solution de l'équation différentielle est donc $f(x)=4e^{2x}$	
	\end{corrBox}
	\item $y'=-3 y$ et $y(2)=1$
	\begin{corrBox}
		L'équation est du type $y'=ay$ avec $a=-3$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{-3x}$ avec C une constante réelle.	
		
		De plus $y(2)=1$ donc $Ce^{-3\times 2}=1$ d'ou $C=\dfrac{1}{e^{-6}}=e^6$.
		
		La solution de l'équation différentielle est donc $f(x)=e^6\times e^{-3x}=e^{6-3x}$	
	\end{corrBox}
\end{enumMet}

		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{\frac{2}{3}x}$ avec C une constante réelle.		
	\end{corrBox}
	\item $y=-2 y^{\prime}$
	\begin{corrBox}
		$y'=-\dfrac{1}{2}y$
		
		L'équation est du type $y'=ay$ avec $a=-\dfrac{1}{2}$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{-\frac{1}{2}x}$ avec C une constante réelle.		
	\end{corrBox}
	\item $4 y^{\prime}-3 y=0$
	\begin{corrBox}
		$4 y^{\prime}-3 y=0 \iff 4y'=3y \iff y'=\dfrac{3}{4}y$
		
		L'équation est du type $y'=ay$ avec $a=\dfrac{3}{4}$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{\frac{3}{4}x}$ avec C une constante réelle.		
	\end{corrBox}
	\item $-5 y^{\prime}+6 y=0$
	\begin{corrBox}
		$-5 y^{\prime}+6 y=0 \iff 5y'=6y \iff y'=\dfrac{6}{5}y$
		
		L'équation est du type $y'=ay$ avec $a=\dfrac{6}{5}$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{\frac{6}{5}x}$ avec C une constante réelle.		
	\end{corrBox}
	\item $2 y^{\prime}-3 y=y^{\prime}+y$
	\begin{corrBox}
		$2 y^{\prime}-3 y=y^{\prime}+y \iff 2y'-y'=y+3y\iff y'=4y$
		
		L'équation est du type $y'=ay$ avec $a=4$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{4x}$ avec C une constante réelle.		
	\end{corrBox}
	\item $2\left(y^{\prime}+2 y\right)=3 y^{\prime}+5 y$
	\begin{corrBox}
		$2\left(y^{\prime}+2 y\right)=3 y^{\prime}+5 y \iff 2y'+4y=3y'+5y\iff 2y'-3y'=5y-4y\iff -y'=y \iff y'=-y$
		
		L'équation est du type $y'=ay$ avec $a=-1$. 
		
		Les solutions de l'équation sont donc les fonctions $f(x)=Ce^{-x}$ avec C une constante réelle.		
	\end{corrBox}
\end{enumMet}
