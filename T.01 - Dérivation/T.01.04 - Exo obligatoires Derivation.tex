	\item $f(x)=\dfrac{1}{3} x^3-6 x+5 \mathrm{e}^x+\dfrac{3}{x}$ sur $\mathbb{R}^*$.
	\item $f(x)=2 \mathrm{e}^{3 x-1}$ sur $\mathbb{R}$.
	\item $f(x)=(5 x+4)^2$ sur $\mathbb{R}$.
	\item $f(x)=\sqrt{1+x^2}$ sur $\mathbb{R}$.
\end{enumMet}
