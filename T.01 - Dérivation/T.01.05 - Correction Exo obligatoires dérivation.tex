Pour chaque fonction $f$, calculer $f^{\prime}(x)$.
%\begin{multicols}{2}
	\begin{enumMet}
	\item $f(x)=\dfrac{1}{3} x^3-6 x+5 \mathrm{e}^x+\dfrac{3}{x}$ sur $\mathbb{R}^*$.
	\begin{corrBox}
		$f'(x)=x^2-6+5e^x-\dfrac{3}{x^2}=\dfrac{x^4-6x^2+5x^2e^x-3}{x^2}$		
	\end{corrBox}
	\item $f(x)=2 \mathrm{e}^{3 x-1}$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f(x)=2e^{u(x)}$ avec $u(x)=3x-1$ donc $u'(x)=3$
		
		$f'(x)=2u'(x)e^{u(x)}=2\times 3\times  \mathrm{e}^{3 x-1}= 6 \mathrm{e}^{3 x-1}$ 		
	\end{corrBox}
	\item $f(x)=(5 x+4)^2$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f(x)=u(x)^2$ avec $u(x)=5x+4$ donc $u'(x)=5$
		
		$f'(x)=2u'(x)u(x)=2\times 5\times (5x+4)=10(5x+4)$		
	\end{corrBox}
	\item $f(x)=\sqrt{1+x^2}$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f(x)=f(u(x))$ avec $u(x)=1+x^2$ donc $u'(x)=2x$
		
		\phantom{$f(x)=f(u(x))$} et $f(X)=\sqrt{X}$ donc $f'(X)=\dfrac{1}{2\sqrt{X}}$
		
		$f'(x)=u'(x)f'(u(x))=2x\times \dfrac{1}{2\sqrt{x^2+1}}$		
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ Dériver avec des fonctions de référence}


\begin{multicols}{2}
	\begin{enumMet}
	\item $f(x)=3 x^2-5 x+4$ pour tout $x$ réel.
	\begin{corrBox}
		$f'(x)=6x-5$		
	\end{corrBox}
	\item $f(x)=\dfrac{5}{3} x^3-\dfrac{1}{2} x^2+8$ pour tout $x$ réel.
	\begin{corrBox}
		$f'(x)=5x^2-x$		
	\end{corrBox}
	\item $f(x)=-x^3+5 x^2+\dfrac{4}{x}$ pour tout $x \neq 0$.
	\begin{corrBox}
		$f'(x)=-3x^2+10x-\dfrac{4}{x^2}=\dfrac{-3x^4+10x^3-4}{x^2}$		
	\end{corrBox}
	\item $f(x)=3 \mathrm{e}^x-5 x^2+4$ pour tout $x$ réel.
	\begin{corrBox}
		$f'(x)=3e^x-10x$		
	\end{corrBox}
	\item $f(x)=(3 x-5)(x+1)$ pour tout $x$ réel.
	\begin{corrBox}
		$f(x)=u(x)\times v(x)$ avec $u(x)=3x-5$ et $v(x)=x+1$
		
		donc $u'(x)=3$ et $v'(x)=1$
		
		d'ou $f'(x)=u'(x)\times v(x)+u(x)\times v'(x)$
		
		$f'(x)=3(x+1)+1(3x-5)=6x-2$		
	\end{corrBox}
	\item $f(x)=\dfrac{4 x}{x^2+1}$ pour tout $x$ réel.
	\begin{corrBox}
		$f(x)=\dfrac{u(x)}{v(x)}$ avec $u(x)=4x$ et $v(x)=x^2+1$
		
		donc $u'(x)=4$ et $v'(x)=2x$
		
		d'ou $f'(x)=\dfrac{u'(x)\times v(x)-u(x)\times v'(x)}{v^2(x)}$
		
		$f'(x)=\dfrac{4(x^2+1)-2x(4x)}{(x^2+1)^2}=\dfrac{-4x^2+4}{(x^2+1)^2}$	
	\end{corrBox}
\end{enumMet}
\end{multicols}

\exerciceDS{ $-$ Dériver $\mathbf{e}^u$}
\begin{enumMet}
	\item $f(x)=\mathrm{e}^{-x}$ pour tout $x$ .
	\begin{corrBox}
		$f'(x)=-e^{-x}$		
	\end{corrBox}
	\item $f(x)=10 \mathrm{e}^{-0,5 x}$ pour tout $x$ .
	\begin{corrBox}
		$f'(x)=10\times -0,5e^{-0,5x}=-5e^{-0,5x}$		
	\end{corrBox}
	\item $f(x)=\dfrac{\mathrm{e}^x}{\mathrm{e}^{-x}+1}$ pour tout $x$ .
	\begin{corrBox}
		$f(x)=\dfrac{u(x)}{v(x)}$ avec $u(x)=e^x$ et $v(x)=e^{-x}+1$
		
		donc $u'(x)=e^x$ et $v'(x)=-e^{-x}$
		
		d'ou $f'(x)=\dfrac{u'(x)\times v(x)-u(x)\times v'(x)}{v^2(x)}$
		
		$f'(x)=\dfrac{e^x(e^{-x}+1)+e^{-x}\times e^x}{(e^{-x}+1)^2}=\dfrac{e^x+2}{(e^{-x}+1)^2}$	
	\end{corrBox}
\end{enumMet}

\begin{enumMet}

\exerciceDS{ $-$ Signe de la dérivée}

\begin{enumMet}
	\item Calculer la dérivée de la fonction $f$ dérivable sur $\mathbb{R}$ définie par $f(x)=x \mathrm{e}^{-0,5 x}$.
	\begin{corrBox}
		$f(x)=u(x)\times v(x)$ avec $u(x)=x$ et $v(x)=\mathrm{e}^{-0,5 x}$
		
		donc $u'(x)=1$ et $v'(x)=-0,5\mathrm{e}^{-0,5 x}$
		
		d'ou $f'(x)=u'(x)\times v(x)+u(x)\times v'(x)$
		
		$f'(x)=1\mathrm{e}^{-0,5 x}-x\times 0,5\mathrm{e}^{-0,5 x}=(1-0,5x)e^{-0,5 x}$		
	\end{corrBox}
	\item Déterminer le signe de $f^{\prime}(x)$.
	\begin{corrBox}
		le signe de $f'(x)$ est celui de $1-0,5x$ car l'exponentielle est toujours positive.
		
		$1-0,5x$ est une fonction affine qui s'annule pour $x=2$ avec $m=-0,5<0$.
		
		$1-0,5x$ et $f'(x)$ sont donc positives sur $]-\infty;2]$ puis négatives ensuite.		
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ Un problème de raccord}

\begin{minipage}[c]{0.48\linewidth}
	Pour terminer une rampe d'accès à un parking, on souhaite raccorder la portion existante, représentée en vert sur le graphique, avec un dernier tronçon, en bleu, allant de $A$ à $B(2 ; 2)$. Les deux tronçons doivent se raccorder parfaitement, sans rupture de pente et la descente démarrer doucement en $A(0 ; 3)$ avec une pente nulle.
L'unité graphique représente 1 m .
\end{minipage} \hfill
\begin{minipage}[c]{0.48\linewidth}
	\includegraphics[scale=0.5]{Img-Exo-obl-01}
\end{minipage}
\begin{enumMet}
	\item Interpréter ces contraintes par des conditions que doit respecter la fonction $f$.
	\begin{corrBox}
		Le tronçon en bleu rejoint le point B, donc $f(2)=2$.
		
		La pente doit être la même en B des 2 côtés donc $f'(2)=g'(2)$.
		
		La pente est nulle au départ donc $f'(0)=0$ et elle démarre du point A donc $f(0)=3$		
	\end{corrBox}
	\item Vérifier que la fonction $f$ définie par $f(x)=-0,25 x^2+3$ remplit toutes ces conditions.
	\begin{corrBox}
		$f'(x)=-0,5x$ et $g'(x)=0,25x-1,5$
		
		$f(2)=-0,25\times 2^2+3=2$
		
		$f'(2)=-0,5\times 2=-1$ et $g'(2)=0,25\times 2-1,5=-1=f'(2)$
		
		$f(0)=-0,25\times 0^2+3=3$ et $f'(0)=-0,5\times 0=0$.
		
		Toutes les conditions sont respectées avec la fonction $f(x)$ proposée. 		
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ Étude d'une fonction polynôme}

Soit $f$ la fonction définie sur $[0 ; 6]$ par $f(x)=-\dfrac{1}{3} x^3+2 x^2-5$.
\begin{enumMet}
	\item Calculer $f^{\prime}(x)$ et étudier son signe.
	\begin{corrBox}
		$f^{\prime}(x)=-x^2+4 x=x(4-x)$. 
		
		Comme $x \geqslant 0, f^{\prime}(x)$, a même signe que $4-x$.
		
		
		Donc $f^{\prime}(x) \geqslant 0$ sur. $[0 ; 4]$ et $f^{\prime}(x) \leqslant 0$ sur. $[4 ; 6]$.		
	\end{corrBox}
	\item Compléter le tableau de variations de $f$ et préciser son maximum.
	\begin{corrBox}
		\begin{tikzpicture}
			\tkzTabInit{$x$ / 1 , $f'(x)$ / 1, $f(x)$ / 1.5}{$0$, $4$, $6$}
			\tkzTabLine{, +, z, -, }
			\tkzTabVar{-/ $-5$, +/ $\dfrac{17}{3}$, -/ $-5$}
		\end{tikzpicture}	
		
		Le maximum de la fonction est donc $\dfrac{17}{3}$ atteint pour $x=4$.	
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ Résistance d'une poutre}

\begin{minipage}[c]{0.74\linewidth}
	Dans un tronc d'arbre on découpe une poutre en forme de parallélépipède rectangle dont une section rectangulaire est représentée ci-contre.
	La résistance à la flexion de cette poutre varie comme le produit $\ell \times h^2$ oú $\ell$ et $h$ sont la longueur et la largeur de la poutre. On prend $\mathrm{OA}=1$ unité de longueur.
\end{minipage} \hfill
\begin{minipage}[c]{0.23\linewidth}
	\includegraphics[scale=0.2]{Img-Exo-obl-02}
\end{minipage}
\begin{enumMet}
	\item Montrer que $h^2=4-\ell^2$,
	\begin{corrBox}
		On utilise le théorème de Pythagore dans le triangle ABC.
		
		$AB^2+BC^2=AC^2 \iff \ell^2+h^2=2^2 \iff h^2=2^2-\ell^2 \iff h^2=4-\ell^2$		
	\end{corrBox}
	\item En déduire que $\ell h^2=f(\ell)$ où $f(x)=-x^3+4 x$ pour $x \geqslant 0$.
	\begin{corrBox}
		$\ell h^2=\ell (4-\ell^2)=-\ell^3+4\ell=f(\ell)$ avec $f(x)=-x^3+4 x$		
	\end{corrBox}
	\item \begin{enumMet}
		\item Compléter le tableau de variations de f sur $[0 ; 2]$.
		\begin{corrBox}
			$f'(x)=-3x^2+4$
			
			$f'(x)=0\iff -3x^2+4=0\iff x^2=\dfrac{4}{3}\iff x=\pm \dfrac{2}{\sqrt{3}}$
			
			\begin{tikzpicture}
				\tkzTabInit{$x$ / 1 , $f'(x)$ / 1}{$-\infty$, $-\dfrac{2}{\sqrt{3}}$, $\dfrac{2}{\sqrt{3}}$, $+\infty$}
				\tkzTabLine{, -, z, +, z,-}
			\end{tikzpicture}
			
			\begin{tikzpicture}
				\tkzTabInit{$x$ / 1 , $f'(x)$ / 1,$f(x)$ / 1.5}{ $0$, $\dfrac{2}{\sqrt{3}}$, $2$}
				\tkzTabLine{, +, z,-}
				\tkzTabVar{-/ , +/ , -/ }
			\end{tikzpicture}		
			
		\end{corrBox}
		\item Comment choisir ( pour que la poutre résiste au mieux à la flexion ?
		
		\begin{corrBox}
			Il faut prendre pour $\ell$ la valeur de $x$ ou on atteint le maximum de $f$, soit $\ell =x=\dfrac{2}{\sqrt{3}}$		
		\end{corrBox}
		
		\item Quel est l'angle $\alpha$ correspondant à $0,1^{\circ}$ près ?
		\begin{corrBox}
			Dans le triangle ABC
			$\cos(\alpha)=\dfrac{AB}{AC}=\dfrac{\ell}{2}$ d'ou $\alpha=\arccos\left(\dfrac{\ell}{2}\right)\approx54,7\degree$		
		\end{corrBox}
		
	\end{enumMet}
\end{enumMet}

\exerciceDS{ $-$ Evolution d'une population}

La population d'une ville est modélisée depuls 2020 par la fonction $f$ donnée par $f(t)=\dfrac{40 t}{t^2+25}+10$, oú $t$ désigne le temps écoulé depuis le 1er janvier 2020, en années, et $f(t)$ la population en dizaines de milliers d'habitants.
\begin{enumMet}
	\item Montrer que $f^{\prime}(t)=40 \dfrac{(5-t)(5+t)}{\left(t^2+25\right)^2}$,
	\begin{corrBox}
		$f(t)=\dfrac{u(t)}{v(t)}$ avec $u(t)=40t$ et $v(t)=t^2+25$
		
		donc $u'(t)=40$ et $v'(t)=2t$
		
		d'ou $f'(x)=\dfrac{u'(x)\times v(x)-u(x)\times v'(x)}{v^2(x)}$
		
		$f'(x)=\dfrac{40(t^2+25)-2t(40t)}{(t^2+25)^2}=\dfrac{40(t^2+25-2t^2}{(t^2+25)^2}=\dfrac{40(25-t^2)}{(t^2+25)^2}=\dfrac{40(5-t)(5+t)}{(t^2+25)^2}=40\dfrac{(5-t)(5+t)}{(t^2+25)^2}$	
	\end{corrBox}
	\item En déduire le tableau de variations de $f$ sur $[0 ; 15]$,
	\begin{corrBox}
		\begin{tikzpicture}
			\tkzTabInit{$x$ / 1 , $f'(x)$ / 1}{$-\infty$, $-5$, $5$, $+\infty$}
			\tkzTabLine{, -, z, +, z,-}
		\end{tikzpicture}
		
		\begin{tikzpicture}
			\tkzTabInit{$x$ / 1 , $f'(x)$ / 1,$f(x)$ / 1.5}{ $0$, $5$, $15$}
			\tkzTabLine{, +, z,-}
			\tkzTabVar{-/$10$ , +/$14$ , -/$12{,}4$ }
		\end{tikzpicture}		
	\end{corrBox}
	\item Quelle sera la population maximale dans ce modèle et en quelle année sera-t-elle atteinte ?
	\begin{corrBox}	
		La population maximale est atteinte pour, $t=5$, donc en 2025, et correspond à 14 dizaines de milliers d'h soit 140 000 habitants.		
	\end{corrBox}
\end{enumMet}
