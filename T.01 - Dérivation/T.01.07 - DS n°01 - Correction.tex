\exerciceDS{ $-$ 5 points}

\begin{enumMet}
	\item Calculer $f^{\prime}(x)$ pour $f(x)=2 x^3-\dfrac{3}{2} x^2+x-1$ sur $\mathbb{R}$. 
	\begin{corrBox}
		$f^{\prime}(x)=6x^2-3x+1$		
	\end{corrBox}
	\item Calculer $f^{\prime}(x)$ pour $f(x)=\dfrac{x+1}{x-4}$ pour $x \neq 4$.
	\begin{corrBox}
		$f$ est du type $\dfrac{u}{v}$ avec $u(x)=x+1$ et $v(x)=x-4$ d'ou $u'(x)=1$ et $v'(x)=1$
		
		$f^{\prime}(x)=\dfrac{u'(x)v(x)-v'(x)u(x)}{v^2(x)}=\dfrac{(x-4)-(x+1)}{(x-4)^2}=\dfrac{-5}{(x-4)^2}$		
	\end{corrBox}
	\item Calculer $f^{\prime}(t)$ pour $f(t)=40 \mathrm{e}^{-0,5 t+2}$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f^{\prime}(x)=40\times (-0,5)\mathrm{e}^{-0,5 t+2}=-20\mathrm{e}^{-0,5 t+2}$		
	\end{corrBox} 
	\item Calculer $f^{\prime}(t)$ pour $f(t)=(2 t-1) \mathrm{e}^{-t}$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f$ est du type $u\times v$ avec $u(t)=2t-1$ et $\mathrm{e}^{-t}$ d'ou $u'(t)=2$ et $v'(t)=-\mathrm{e}^{-t}$
		
		$f^{\prime}(t)=u'(x)v(x)+v'(x)u(x)=2\mathrm{e}^{-t}-(2t-1)\mathrm{e}^{-t}=(-2t+3)\mathrm{e}^{-t}$		
	\end{corrBox}
	\item Calculer $f^{\prime}(t)$ pour $f(t)=\left(2 t^2-3 t+1\right)^2$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f$ est du type $f(u(t))$ avec $u(t)=2t^2-3t+1$ et $f(X)=X^2$ d'ou $u'(t)=4t-3$ et $f'(X)=2X$ 
		
		$f^{\prime}(x)=2(4t-3)(2t^2-3t+1)$		
	\end{corrBox}
	\item Calculer $f^{\prime}(x)$ pour $f(x)=\sqrt{4-x^2}$ sur [-2; 2].
	\begin{corrBox}
		$f$ est du type $f(u(x))$ avec $u(x)=4-x^2$ et $f(X)=\sqrt{X}$ d'ou $u'(x)=-2x$ et $f'(X)=\dfrac{1}{2\sqrt{X}}$ 
		
		$f^{\prime}(x)=\dfrac{-2x}{2\sqrt{4-x^2}}=-\dfrac{x}{\sqrt{4-x^2}}$		
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ 4 points}

Une personne a consommé de l'alcool au cours d'un repas. Le taux d'alcoolémie dans son organisme, en grammes par litre de sang, est donné par $A(t)=5 t e^{-t}$ avec $0 \leqslant t \leqslant 6$, où $t$ est le temps écoulé depuis la fin du repas, en heures.
\begin{enumMet}
	\item Déterminer $A^{\prime}(t)$.
	\begin{corrBox}
		$A$ est du type $u\times v$ avec $u(t)=5t$ et $\mathrm{e}^{-t}$ d'ou $u'(t)=5$ et $v'(t)=-\mathrm{e}^{-t}$
		
		$f^{\prime}(t)=u'(x)v(x)+v'(x)u(x)=5\mathrm{e}^{-t}-5t\mathrm{e}^{-t}=5(1-t)\mathrm{e}^{-t}$		
	\end{corrBox}
	\item Dresser le tableau de signes de la dérivée sur \R
	\begin{corrBox}
		\begin{tikzpicture}
			\tkzTabInit{$x$ / 1 , $A'(t)$ / 1}{$-\infty$, $1$, $+\infty$}
			\tkzTabLine{, +, z, -, }
		\end{tikzpicture}
	\end{corrBox}
	\item en déduire le tableau de variation de la fonction $A(t)$ sur $[0;6]$
	\begin{corrBox}
		\begin{tikzpicture}
			\tkzTab{$x$ / 1 , $A'(t)$ / 1, $A(t)$ / 1.5}{$0$, $1$, $6$}
			{, +, z, -, }
			{-/, +/ $A(1)\approx1{,}84$, -/}
		\end{tikzpicture}		
	\end{corrBox}
	\item préciser le maximum de la fonction et au bout de combien de temps est-il atteint?
	\begin{corrBox}
		Le maximum de la fonction est environ 1,84 g/L de sang et il est atteint au bout d'une heure		
	\end{corrBox}
\end{enumMet}

		$f^{\prime}(x)=u'(x)v(x)+v'(x)u(x)=(-2x+1)\mathrm{e}^{x}+(-x^2+x+1)\mathrm{e}^{x}=(-x^2-x+2)\mathrm{e}^{x}$		
	\end{corrBox}

		\begin{tikzpicture}
