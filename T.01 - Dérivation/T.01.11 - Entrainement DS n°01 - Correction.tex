\exerciceDS{ $-$ 5 points}

\begin{enumMet}
	\item Calculer $f^{\prime}(x)$ pour $f(x)=\dfrac{1}{3} x^3-\dfrac{5}{2} x^2+4 x-7$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f^{\prime}(x)=x^2-5x+4$
	\end{corrBox}
	\item Calculer $f^{\prime}(x)$ pour $f(x)=\dfrac{2x-3}{x+1}$ pour $x \neq -1$.
	\begin{corrBox}
		$f$ est du type $\dfrac{u}{v}$ avec $u(x)=2x-3$ et $v(x)=x+1$ d'ou $u'(x)=2$ et $v'(x)=1$

		$f^{\prime}(x)=\dfrac{u'(x)v(x)-v'(x)u(x)}{v^2(x)}=\dfrac{2(x+1)-(2x-3)}{(x+1)^2}=\dfrac{5}{(x+1)^2}$
	\end{corrBox}
	\item Calculer $f^{\prime}(t)$ pour $f(t)=25 \mathrm{e}^{-0,2 t+1}$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f^{\prime}(t)=25\times (-0,2)\mathrm{e}^{-0,2 t+1}=-5\mathrm{e}^{-0,2 t+1}$
	\end{corrBox}
	\item Calculer $f^{\prime}(t)$ pour $f(t)=(3 t+2) \mathrm{e}^{-2t}$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f$ est du type $u\times v$ avec $u(t)=3t+2$ et $v(t)=\mathrm{e}^{-2t}$ d'ou $u'(t)=3$ et $v'(t)=-2\mathrm{e}^{-2t}$

		$f^{\prime}(t)=u'(t)v(t)+v'(t)u(t)=3\mathrm{e}^{-2t}-2(3t+2)\mathrm{e}^{-2t}=(-6t-1)\mathrm{e}^{-2t}$
	\end{corrBox}
	\item Calculer $f^{\prime}(t)$ pour $f(t)=\left(3 t^2+2 t-5\right)^2$ sur $\mathbb{R}$.
	\begin{corrBox}
		$f$ est du type $f(u(t))$ avec $u(t)=3t^2+2t-5$ et $f(X)=X^2$ d'ou $u'(t)=6t+2$ et $f'(X)=2X$

		$f^{\prime}(t)=2(6t+2)(3t^2+2t-5)=4(3t+1)(3t^2+2t-5)$
	\end{corrBox}
	\item Calculer $f^{\prime}(x)$ pour $f(x)=\sqrt{9-x^2}$ sur ]-3; 3[.
	\begin{corrBox}
		$f$ est du type $f(u(x))$ avec $u(x)=9-x^2$ et $f(X)=\sqrt{X}$ d'ou $u'(x)=-2x$ et $f'(X)=\dfrac{1}{2\sqrt{X}}$

		$f^{\prime}(x)=\dfrac{-2x}{2\sqrt{9-x^2}}=-\dfrac{x}{\sqrt{9-x^2}}$
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ 4 points}

Un circuit électrique est formé d'un générateur de force électromotrice $E=20,0 \mathrm{~V}$ (Volts), de résistance interne $\mathrm{r}=4,0 \Omega$ et d'une résistance variable $R$, en Ohms ( $\Omega$ ).

La puissance absorbée par la résistance $R$ est $\dfrac{400 R}{(R+4)^2}$, en Watts.

\begin{enumMet}
	\item Soit $P(R)=\dfrac{400 R}{(R+4)^2}$.
	\begin{enumMet}
		\item Calculer $P'(R)$, la dérivée de $P(R)$ sur $[0 ;+\infty[$.
		\begin{corrBox}
			$P$ est du type $\dfrac{u}{v}$ avec $u(R)=400R$ et $v(R)=(R+4)^2$ d'ou $u'(R)=400$ et $v'(R)=2(R+4)$

			$P^{\prime}(R)=\dfrac{u'(R)v(R)-v'(R)u(R)}{v^2(R)}=\dfrac{400(R+4)^2-400R\times 2(R+4)}{(R+4)^4}$

			$P'(R)=\dfrac{400R^2+3200R+6400-800R^2-3200R}{(R+4)^4}=\dfrac{6400-400R^2}{(R+4)^4}=\dfrac{400(16-R^2)}{(R+4)^4}$

			$P'(R)=\dfrac{400(4-R)(4+R)}{(R+4)^4}=\dfrac{400(4-R)}{(R+4)^3}$
		\end{corrBox}
		\item Dresser le tableau de signes de $P'(R)$ sur $[0 ;+\infty[$.
		\begin{corrBox}
			Sur $[0;+\infty[$, $(R+4)^3>0$ donc $P'(R)$ est du signe de $4-R$.

			\begin{tikzpicture}
				\tkzTabInit{$R$ / 1 , $P'(R)$ / 1}{$0$, $4$, $+\infty$}
				\tkzTabLine{, +, z, -, }
			\end{tikzpicture}
		\end{corrBox}
		\item En déduire le tableau de variation de $P$ sur $[0 ;+\infty[$.
		\begin{corrBox}
			\begin{tikzpicture}
				\tkzTab{$R$ / 1 , $P'(R)$ / 1, $P(R)$ / 1.5}{$0$, $4$, $+\infty$}
				{, +, z, -, }
				{-/ 0, +/ $P(4)=25$, -/ 0}
			\end{tikzpicture}
		\end{corrBox}
	\end{enumMet}
	\item En déduire la valeur de $R$ pour laquelle la puissance est maximale et préciser cette puissance.
	\begin{corrBox}
		La puissance est maximale quand $R=4\ \Omega$ et vaut alors $P=\dfrac{400\times 4}{(4+4)^2}=\dfrac{1600}{64}=25$ W
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ 4 points}

Un patient reçoit une injection de médicament. La concentration du médicament dans le sang, en milligrammes par litre, est donnée par $C(t)=4 t e^{-0,5 t}$ avec $0 \leqslant t \leqslant 10$, où $t$ est le temps écoulé depuis l'injection, en heures.
\begin{enumMet}
	\item Déterminer $C^{\prime}(t)$.
	\begin{corrBox}
		$C$ est du type $u\times v$ avec $u(t)=4t$ et $v(t)=\mathrm{e}^{-0,5t}$ d'ou $u'(t)=4$ et $v'(t)=-0,5\mathrm{e}^{-0,5t}$

		$C^{\prime}(t)=u'(t)v(t)+v'(t)u(t)=4\mathrm{e}^{-0,5t}-2t\mathrm{e}^{-0,5t}=(4-2t)\mathrm{e}^{-0,5t}=2(2-t)\mathrm{e}^{-0,5t}$
	\end{corrBox}
	\item Dresser le tableau de signes de la dérivée sur \R
	\begin{corrBox}
		Pour tout réel $t$, $\mathrm{e}^{-0,5t}>0$ donc $C'(t)$ est du signe de $2-t$.

		\begin{tikzpicture}
			\tkzTabInit{$t$ / 1 , $C'(t)$ / 1}{$-\infty$, $2$, $+\infty$}
			\tkzTabLine{, +, z, -, }
		\end{tikzpicture}
	\end{corrBox}
	\item en déduire le tableau de variation de la fonction $C(t)$ sur $[0;10]$
	\begin{corrBox}
		\begin{tikzpicture}
			\tkzTab{$t$ / 1 , $C'(t)$ / 1, $C(t)$ / 1.5}{$0$, $2$, $10$}
			{, +, z, -, }
			{-/ $0$, +/ $C(2)\approx2{,}94$, -/ $C(10)\approx0{,}27$}
		\end{tikzpicture}
	\end{corrBox}
	\item préciser le maximum de la fonction et au bout de combien de temps est-il atteint ?
	\begin{corrBox}
		Le maximum de la fonction est $C(2)=8\mathrm{e}^{-1}\approx 2{,}94$ mg/L de sang et il est atteint au bout de 2 heures
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ 4 points}

On considère la fonction $f: x \mapsto e^{x^2+3 x-4}$, définie sur $\mathbb{R}$
\begin{enumMet}
	\item Calculer $f^{\prime}(x)$ pour tout réel $x$.
	\begin{corrBox}
		$f$ est du type $f(u(x))$ avec $u(x)=x^2+3 x-4$ et $f(X)=e^{X}$ d'ou $u'(x)=2x+3$ et $f'(X)=e^{X}$

		$f^{\prime}(x)=(2x+3)e^{x^2+3 x-4}$
	\end{corrBox}
	\item Dresser le tableau de signes de la dérivée sur \R
	\begin{corrBox}
		Pour tout réel $x$, $e^{x^2+3x-4}>0$ donc $f'(x)$ est du signe de $2x+3$.

	\item En déduire le tableau de variations de $f$.
