\exerciceDS{ $-$ (3 points)}

On considère la fonction $f: x \mapsto \begin{cases}2 x+5 & \text { si } x \leqslant-1 \\ -2 x+1 & \text { si }-1<x<2  \\ 3x-3 & \text { si } x \geqslant 2\end{cases}$

La fonction $f$ est-elle continue en $-1$? et en $2$?

\begin{corrBox}
	Étudions alors la continuité de \(f\) en $-1$ et en $2$ :
	
	\begin{itemMet}
		\item \(\lim _{x \rightarrow -1^{-}} f(x)=\lim _{x \rightarrow -1^{-}}2x+5=2\times (-1)+5=3\)\\
		\(\lim _{x \rightarrow -1^{+}} f(x)=\lim _{x \rightarrow -1^{+}} -2x+1=-2\times (-1)+1=3\)\\
		Donc : \(\lim _{x \rightarrow -1^{-}} f(x)=\lim _{x \rightarrow -1^{+}} f(x)=f(-1)\)\\
		La fonction \(f\) est continue en $-1$ .
		\item \(\lim _{x \rightarrow 2^{-}} f(x)=\lim _{x \rightarrow 2^{-}} -2x+1=-2\times 2+1=-3\)\\
		\(\lim _{x \rightarrow 2^{+}} f(x)=\lim _{x \rightarrow 2^{+}}3x-3=3 \times 6-3=3\neq\lim _{x \rightarrow 2^{-}} f(x)\)\\
		La fonction \(f\) n'est donc pas continue en 2 .\\
	\end{itemMet}	
\end{corrBox}

\exerciceDS{ \ding{46} $-$ (4 points)}

Pour tout réel $x$, on pose $f(x)=x^4-3 x-5$.
\begin{enumMet}
	\item \ding{46} Calculer $f(-2)$ et $f(1)$
	\begin{corrBox}
		$f(-2)=(-2)^4-3\times (-2)-5=17$ et $f(1)=1^4-3\times 1-5=-7$		
	\end{corrBox}
	\item \ding{46} En déduire que l'équation $x^4=3 x+5$ possède au moins une solution sur $[-2\ ;\ 1]$.
	\begin{corrBox}
		La fonction $f$ étant continue sur $[-2\ ;\ 1]$, d'après le théorème des valeurs intermédiaires, il existe un réel $c \in[-2\ ;\ 1]$ tel que $f(c)=0$, soit $c^4-3 c-5=0$ ou encore $c^4=3c+5$.		
	\end{corrBox}
	\item \ding{46} Donner une solution de cette équation au centième près.
	\begin{corrBox}
		En utilisant un algorithme de balayage, on trouve que $c \approx -1,13$		
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ (8 points)}

On considère la fonction $f: x \mapsto 2e^x-15 x$, définie sur $\mathbb{R}$.
\begin{enumMet}
	\item Justifier que $f$ est continue et dérivable sur $\mathbb{R}$ et déterminer $f^{\prime}(x)$ pour tout réel $x$
	\begin{corrBox}
		$f$ est dérivable comme somme de fonctions dérivables (et est donc continue). Pour tout réel $x$, $f^{\prime}(x)=2e^x-15$		
	\end{corrBox}
	\item Quel est le sens de variation de $f$ sur l'intervalle $[0 ; 2]$
	\begin{corrBox}
		 $f'(x)$ est croissante $[0 ; 2]$ (fonction exponentielle) et $f'(2)<0$. $f'(x)<0$ sur $[0 ; 2]$	donc $f$ est décroissante sur $[0 ; 2]$	
	\end{corrBox}
	\item Que vaut $f(0)$ ? Quel est le signe de $f(2)$ ?
	\begin{corrBox}
		$f(0)=2e^0-15\times 0=2>0$ et $f(2)\approx -15,2<0$		
	\end{corrBox}
	\item En déduire que l'équation $2e^x=15x$ admet exactement une solution sur $[0 ; 2]$.
	\begin{corrBox}
		$f(0)=2>0$ et $f(2)<0$. Résoudre $2e^x=15x$ revient à résoudre $f(x)=0$. Or, $f$ est continue sur $[0 ; 2], f(0)>0$ et $f(2)<0$. D'après le théorème des valeurs intermédiaires, il existe un réel $c$ dans $[0,2]$ tel que $f(c)=0$. De plus, la fonction $f$ étant strictement décroissante sur cet intervalle, le réel $c$ est unique.		
	\end{corrBox}
	\item Donner une solution de cette équation au centième près.
	\begin{corrBox}
		En utilisant un algorithme de balayage, on trouve que $c \approx 0,16$

\end{enumMet}

$$
g(x)=\frac{1}{3} x^3-5x^2+9x+5
$$

\begin{enumMet}
	\item Dresser le tableau de variations de la fonction $g$ sur \R.
	\begin{corrBox}

	\item Montrer que l'équation $g(x)=0$ admet une solution unique $\alpha$ dans $[0\ ;\ 8]$. Déterminer un encadrement de $\alpha$ au millième près.
	\begin{corrBox}
		$g$ est continue sur $[0\ ;\ 8]$ car une fonction polynôme est continue sur \R.

\end{enumMet}
