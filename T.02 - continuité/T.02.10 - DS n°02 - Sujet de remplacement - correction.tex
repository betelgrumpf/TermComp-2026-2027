\exerciceDS{ $-$ (3 points)}

On considère la fonction $f: x \mapsto \begin{cases}3 x+4 & \text { si } x \leqslant-2 \\ -x-4 & \text { si }-2<x<1  \\ 2x-1 & \text { si } x \geqslant 1\end{cases}$

La fonction $f$ est-elle continue en $-2$ ? et en $1$ ?

\begin{corrBox}
	Étudions la continuité de \(f\) en $-2$ et en $1$ :

	\begin{itemMet}
		\item \(\lim \limits_{x \rightarrow -2^{-}} f(x)=\lim \limits_{x \rightarrow -2^{-}}3x+4=3\times (-2)+4=-2\)\\
		\(\lim \limits_{x \rightarrow -2^{+}} f(x)=\lim \limits_{x \rightarrow -2^{+}} -x-4=-(-2)-4=-2\)\\
		Donc : \(\lim \limits_{x \rightarrow -2^{-}} f(x)=\lim \limits_{x \rightarrow -2^{+}} f(x)=f(-2)\)\\
		La fonction \(f\) est continue en $-2$.
		\item \(\lim \limits_{x \rightarrow 1^{-}} f(x)=\lim \limits_{x \rightarrow 1^{-}} -x-4=-1-4=-5\)\\
		\(\lim \limits_{x \rightarrow 1^{+}} f(x)=\lim \limits_{x \rightarrow 1^{+}}2x-1=2 \times 1-1=1\neq\lim \limits_{x \rightarrow 1^{-}} f(x)\)\\
		La fonction \(f\) n'est donc pas continue en $1$.\\
	\end{itemMet}
\end{corrBox}

\exerciceDS{ \ding{46} $-$ (4 points)}

Pour tout réel $x$, on pose $f(x)=x^4-2 x-6$.
\begin{enumMet}
	\item \ding{46} Calculer $f(-2)$ et $f(0)$
	\begin{corrBox}
		$f(-2)=(-2)^4-2\times (-2)-6=14$ et $f(0)=0^4-2\times 0-6=-6$
	\end{corrBox}
	\item \ding{46} En déduire que l'équation $x^4=2 x+6$ possède au moins une solution sur $[-2\ ;\ 0]$.
	\begin{corrBox}
		La fonction $f$ est continue sur $[-2\ ;\ 0]$ (fonction polynôme), $f(-2)>0$ et $f(0)<0$. D'après le théorème des valeurs intermédiaires, il existe un réel $c \in[-2\ ;\ 0]$ tel que $f(c)=0$, soit $c^4-2 c-6=0$ ou encore $c^4=2c+6$.
	\end{corrBox}
	\item \ding{46} Donner une solution de cette équation au centième près.
	\begin{corrBox}
		En utilisant un algorithme de balayage, on trouve que $c \approx -1,35$ \quad ($f(-1,35)\approx 0,02>0$ et $f(-1,34)\approx -0,10<0$)
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ (8 points)}

On considère la fonction $f: x \mapsto e^x-8 x$, définie sur $\mathbb{R}$.
\begin{enumMet}
	\item Justifier que $f$ est continue et dérivable sur $\mathbb{R}$ et déterminer $f^{\prime}(x)$ pour tout réel $x$
	\begin{corrBox}
		$f$ est dérivable sur $\mathbb{R}$ comme somme de fonctions dérivables (et est donc continue). Pour tout réel $x$, $f^{\prime}(x)=e^x-8$
	\end{corrBox}
	\item Quel est le sens de variation de $f$ sur l'intervalle $[0 ; 2]$ ?
	\begin{corrBox}
		Si $0 \leqslant x \leqslant 2$, par croissance de la fonction exponentielle, $e^x \leqslant e^2$ donc $f'(x)=e^x-8\leqslant e^2-8\approx -0,61<0$.\\
		$f'(x)<0$ sur $[0 ; 2]$ donc $f$ est strictement décroissante sur $[0 ; 2]$.
	\end{corrBox}
	\item Que vaut $f(0)$ ? Quel est le signe de $f(2)$ ?
	\begin{corrBox}
		$f(0)=e^0-8\times 0=1>0$ et $f(2)=e^2-16\approx -8,61<0$
	\end{corrBox}
	\item En déduire que l'équation $e^x=8 x$ admet exactement une solution sur $[0 ; 2]$.
	\begin{corrBox}
		Résoudre $e^x=8x$ revient à résoudre $f(x)=0$. Or, $f$ est continue sur $[0 ; 2]$, $f(0)>0$ et $f(2)<0$. D'après le théorème des valeurs intermédiaires, il existe un réel $c$ dans $[0 ; 2]$ tel que $f(c)=0$. De plus, la fonction $f$ étant strictement décroissante sur cet intervalle, le réel $c$ est unique.
	\end{corrBox}
	\item Donner une solution de cette équation au centième près.
	\begin{corrBox}
		En utilisant un algorithme de balayage, on trouve que $c \approx 0,14$ \quad ($f(0,14)\approx 0,030>0$ et $f(0,15)\approx -0,038<0$)
	\end{corrBox}
\end{enumMet}

\exerciceDS{ $-$ (6 points)}

Soit $g$ la fonction définie sur $[0\ ;\ 6]$ par :

$$
g(x)=\frac{1}{3} x^3-4x^2+7x+4
$$

\begin{enumMet}
	\item Dresser le tableau de variations de la fonction $g$ sur \R.
	\begin{corrBox}
		$g'(x)=x^2-8x+7$\quad \quad \quad \quad $\Delta=36$\quad \quad \quad \quad $x_1=1$\quad \quad \quad \quad $x_2=7$

		$g(1)=\dfrac{1}{3}-4+7+4=\dfrac{22}{3}$ \quad et \quad $g(7)=\dfrac{343}{3}-196+49+4=-\dfrac{86}{3}$

		\begin{tikzpicture}
			\tkzTab{$x$ / 1 , $g'(x)$ / 1, $g(x)$ / 1.5}{$-\infty$, $1$,$7$, $+\infty$}
			{, +, z, -, z,+,}
			{-/ , +/ $\dfrac{22}{3}$, -/ $-\dfrac{86}{3}$, +/}
		\end{tikzpicture}
	\end{corrBox}
	\item Montrer que l'équation $g(x)=0$ admet une solution unique $\alpha$ dans $[0\ ;\ 6]$. Déterminer un encadrement de $\alpha$ au millième près.
	\begin{corrBox}
		$g$ est continue sur $[0\ ;\ 6]$ car une fonction polynôme est continue sur \R.

		Sur $[0\ ;\ 1]$ la fonction est strictement croissante et $g(0)=4>0$, donc $g(x)\geqslant 4>0$. Il n'y a donc pas de solution à l'équation $g(x)=0$ sur cet intervalle.

		Sur $[1\ ;\ 6]$, la fonction est strictement décroissante, $g(1)=\dfrac{22}{3}>0$ et $g(6)=72-144+42+4=-26<0$, donc d'après le TVI (cas strictement monotone), il existe une solution unique $\alpha$ à l'équation $g(x)=0$.

		À l'aide de la calculatrice, on trouve $2,740<\alpha<2,741$
	\end{corrBox}
	\item En déduire le tableau de signes de $g$ sur $[0\ ;\ 6]$.

	\begin{corrBox}
		\begin{tikzpicture}
			\tkzTabInit{$x$ / 1 , $g(x)$ / 1}{$0$, $\alpha$, $6$}
			\tkzTabLine{, +, z, -, }
		\end{tikzpicture}
	\end{corrBox}
\end{enumMet}
